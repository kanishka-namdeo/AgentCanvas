   431→  1629→function buildUserFlow(flow: string, ox: number, oy: number): UserFlowResult {
   432→  1630→  // Each step is a mobile screen (375 wide) with a 64px gap.
   433→  1631→  const SCREEN_W = 375;
   434→  1632→  const GAP = 80;
   435→  1633→  const ARROW_COLOR = '#94a3b8';
   436→  1634→
   437→  1635→  const flows: Record<string, string[]> = {
   438→  1636→    onboarding: ['mobile_login', 'mobile_signup', 'mobile_dashboard'],
   439→  1637→    ecommerce: ['web_landing', 'mobile_dashboard', 'mobile_dashboard', 'mobile_login'],
   440→  1638→    auth: ['mobile_login', 'mobile_dashboard', 'mobile_dashboard'],
   441→  1639→    signup_funnel: ['web_landing', 'mobile_signup', 'mobile_login', 'mobile_dashboard'],
   442→  1640→  };
   443→  1641→  const screens = flows[flow] ?? flows.onboarding;
   444→  1642→
   445→  1643→  const frameIds: string[] = [];
   446→  1644→  const shapes: Array<Partial<Shape> & { id: string }> = [];
   447→  1645→
   448→  1646→  screens.forEach((tmpl, i) => {
   449→  1647→    const sx = ox + i * (SCREEN_W + GAP);

   455→  1653→
   456→  1654→    // Add an arrow connector to the next screen.
   457→  1655→    if (i < screens.length - 1) {
   458→  1656→      const arrowStartX = sx + SCREEN_W;

   485→  1683→function buildDiagram(template: string, labels: string[], ox: number, oy: number): DiagramResult {
   486→  1684→  const shapes: Array<Partial<Shape> & { id: string }> = [];
   487→  1685→  const NODE_W = 160;
   488→  1686→  const NODE_H = 56;
   489→  1687→  const NODE_FILL = '#ffffff';
   490→  1688→  const NODE_STROKE = '#475569';
   491→  1689→  const ARROW = '#94a3b8';
   492→  1690→
   493→  1691→  const addNode = (label: string, x: number, y: number) => {
   494→  1692→    shapes.push({
   495→  1693→      id: crypto.randomUUID(),
   496→  1694→      type: 'rectangle',
   497→  1695→      name: label,
   498→  1696→      x, y, width: NODE_W, height: NODE_H,
   499→  1697→      fill: NODE_FILL, stroke: NODE_STROKE, strokeWidth: 1, radius: 8,
   500→  1698→      fontSize: 13, textColor: '#0f172a',
   501→  1699→    });
   502→  1700→    shapes.push({
   503→  1701→      id: crypto.randomUUID(),
   504→  1702→      type: 'text',
   505→  1703→      name: `${label} label`,
   506→  1704→      x: x + 12, y: y + 18, width: NODE_W - 24, height: 20,
   507→  1705→      fill: 'transparent', text: label, fontSize: 13, textColor: '#0f172a',
   508→  1706→      stroke: 'transparent', strokeWidth: 0, radius: 0,
   509→  1707→    });
   510→  1708→  };
   511→  1709→  const addArrow = (x1: number, y1: number, x2: number, y2: number) => {
   512→  1710→    shapes.push({
   513→  1711→      id: crypto.randomUUID(),
   514→  1712→      type: 'line',
   515→  1713→      name: 'connector',
   516→  1714→      x: x1, y: y1, width: x2 - x1, height: y2 - y1,
   517→  1715→      fill: ARROW, stroke: ARROW, strokeWidth: 2, radius: 0,
   518→  1716→      fontSize: 14, textColor: ARROW,
   519→  1717→    });
   520→  1718→  };
   521→  1719→
   522→  1720→  if (template === 'flowchart') {
   523→  1721→    // Top-down: each node centered, vertical arrow between.
   524→  1722→    const cx = ox + 200;
   525→  1723→    let cy = oy;
   526→  1724→    const STEP = NODE_H + 60;
   527→  1725→    labels.forEach((label, i) => {
   528→  1726→      addNode(label, cx, cy);
   529→  1727→      if (i < labels.length - 1) {
   530→  1728→        addArrow(cx + NODE_W / 2, cy + NODE_H, cx + NODE_W / 2, cy + STEP);
   531→  1729→      }
   532→  1730→      cy += STEP;
   533→  1731→    });
   534→  1732→  } else {
   535→  1733→    // Mindmap: central node + radial children.
   536→  1734→    const cx = ox + 360;
   537→  1735→    const cy = oy + 280;
   538→  1736→    const centerLabel = labels[0];
   539→  1737→    addNode(centerLabel, cx - NODE_W / 2, cy - NODE_H / 2);
   540→  1738→    const radius = 220;
   541→  1739→    const childCount = labels.length - 1;
   542→  1740→    for (let i = 1; i < labels.length; i++) {
   543→  1741→      const angle = (i - 1) / Math.max(1, childCount) * Math.PI * 2;
   544→  1742→      const px = cx + Math.cos(angle) * radius - NODE_W / 2;
   545→  1743→      const py = cy + Math.sin(angle) * radius - NODE_H / 2;
   546→  1744→      addArrow(cx, cy, px + NODE_W / 2, py + NODE_H / 2);
   547→  1745→      addNode(labels[i], px, py);
   548→  1746→    }
   549→  1747→  }
   550→  1748→  return { shapes };
   551→  1749→}
