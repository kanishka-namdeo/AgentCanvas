    61→// ---- Tool execution context -------------------------------------------------
    62→//
    63→// `defineTool`'s `execute` signature is:
    64→//   (toolCallId, params, signal, onUpdate, ctx) => Promise<AgentToolResult>
    65→//
    66→// The tools here need access to:
    67→//   1. The current canvas state (to read shapes / find by id).
    68→//   2. A patch sink — every mutation is recorded as a `CanvasPatch` so it
    69→//      can be persisted, broadcast to viewers, and replayed.
    70→//
    71→// We don't want to rewire the SDK's call signature, so we attach these via
    72→// a closure when the tools are constructed for a given session. The factory
    73→// `createCanvasTools()` below returns an array of `ToolDefinition`s bound
    74→// to a specific session context.
    75→
    76→export interface CanvasToolContext {
    77→  /// Read-only snapshot of the current canvas state. The runner refreshes
    78→  /// this before each tool call so the agent always sees fresh data.
    79→  getShapes: () => Shape[];
    80→  /// Read-only snapshot of the current design tokens.
    81→  getTokens: () => DesignTokens;
    82→  /// Apply a patch (the tool's effect) and return the patched document.
    83→  applyPatch: (patch: CanvasPatch) => CanvasPatch;
    84→}

   118→// ---- Helpers ----------------------------------------------------------------
   119→
   120→/// Coerce LLM-provided arguments into the types the schema expects.
   121→/// LLMs sometimes pass numbers as strings (e.g. `x: "400"` instead of `400`).
   122→/// This helper normalizes those before they reach the patch layer.
   123→function coerceShapeInput(params: Static<typeof ShapeInputSchema>): Partial<Shape> {
   124→  const out: Partial<Shape> = { type: params.type as Shape['type'] };
   125→  if (params.name !== undefined) out.name = String(params.name);
   126→  if (params.x !== undefined) out.x = Number(params.x) || 0;
   127→  if (params.y !== undefined) out.y = Number(params.y) || 0;
   128→  if (params.width !== undefined) out.width = Number(params.width) || 0;
   129→  if (params.height !== undefined) out.height = Number(params.height) || 0;
   130→  if (params.rotation !== undefined) out.rotation = Number(params.rotation) || 0;
   131→  if (params.opacity !== undefined) out.opacity = Math.max(0, Math.min(1, Number(params.opacity) || 1));
   132→  if (params.fill !== undefined) out.fill = String(params.fill);
   133→  if (params.stroke !== undefined) out.stroke = String(params.stroke);
   134→  if (params.strokeWidth !== undefined) out.strokeWidth = Number(params.strokeWidth) || 0;
   135→  if (params.radius !== undefined) out.radius = Number(params.radius) || 0;
   136→  if (params.text !== undefined) out.text = String(params.text);
   137→  if (params.fontSize !== undefined) out.fontSize = Number(params.fontSize) || 16;
   138→  if (params.textColor !== undefined) out.textColor = String(params.textColor);
   139→  return out;
   140→}

   142→/// Convert HSL → hex. Used by the palette generator.
   143→function hslToHex(h: number, s: number, l: number): string {
   144→  s /= 100; l /= 100;
   145→  const k = (n: number) => (n + h / 30) % 12;
   146→  const a = s * Math.min(l, 1 - l);
   147→  const f = (n: number) => {
   148→    const v = l - a * Math.max(-1, Math.min(k(n) - 3, Math.min(9 - k(n), 1)));
   149→    return Math.round(v * 255).toString(16).padStart(2, '0');
   150→  };
   151→  return `#${f(0)}${f(8)}${f(4)}`;
   152→}
   153→
   154→/// Parse a hex color → {h, s, l}.
   155→function hexToHsl(hex: string): { h: number; s: number; l: number } {
   156→  const m = hex.replace('#', '');
   157→  const r = parseInt(m.slice(0, 2), 16) / 255;
   158→  const g = parseInt(m.slice(2, 4), 16) / 255;
   159→  const b = parseInt(m.slice(4, 6), 16) / 255;
   160→  const max = Math.max(r, g, b);
   161→  const min = Math.min(r, g, b);
   162→  let h = 0;
   163→  let s = 0;
   164→  const l = (max + min) / 2;
   165→  const d = max - min;
   166→  if (d !== 0) {
   167→    s = l > 0.5 ? d / (2 - max - min) : d / (max + min);
   168→    switch (max) {
   169→      case r: h = (g - b) / d + (g < b ? 6 : 0); break;
   170→      case g: h = (b - r) / d + 2; break;
   171→      case b: h = (r - g) / d + 4; break;
   172→    }
   173→    h *= 60;
   174→  }
   175→  return { h, s: s * 100, l: l * 100 };
   176→}

   190→      'Returns the new shape id. The shape appears immediately on every viewer\'s screen.',

   207→      ctx.applyPatch(patch);
   208→      return {
   209→        content: [
   210→          {
   211→            type: 'text',
   212→            text: `Created ${params.type} with id ${id}. Coordinates: (${coerced.x ?? 0}, ${coerced.y ?? 0}), size ${coerced.width ?? 100}×${coerced.height ?? 100}.`,

  1357→// ---- Convert Pi tools → OpenAI tool spec -----------------------------------
  1358→//
  1359→// The Pi Agent SDK normally drives tools through `pi-ai` provider adapters.
  1360→// In this sandbox we don't have Anthropic/OpenAI keys, so we run the agent
  1361→// via `z-ai-web-dev-sdk` (which speaks the OpenAI tool-calling format).
  1362→// This helper converts our TypeBox-defined Pi tools into the JSON-schema
  1363→// tool spec that the LLM expects.
  1364→
  1365→import { Value } from '@sinclair/typebox/value';
  1366→
  1367→export interface OpenAIToolSpec {
  1368→  type: 'function';
  1369→  function: {
  1370→    name: string;
  1371→    description: string;
  1372→    parameters: object;
  1373→  };
  1374→}
  1375→
  1376→export function toolsToOpenAISpec(tools: ReturnType<typeof createCanvasTools>): OpenAIToolSpec[] {
  1377→  return tools.map((t) => ({
  1378→    type: 'function',
  1379→    function: {
  1380→      name: t.name,
  1381→      description: t.description,
  1382→      // TypeBox schemas ARE JSON-schema, so we can pass them through after
  1383→      // cleaning up any TypeBox-specific symbols.
  1384→      parameters: Value.Clean(t.parameters, {}) as object,
  1385→    },
  1386→  }));
  1387→}
  1388→
  1389→// ---- Execute a tool by name -------------------------------------------------
  1390→
  1391→export async function executeTool(
  1392→  tools: ReturnType<typeof createCanvasTools>,
  1393→  toolName: string,
  1394→  args: any,
  1395→): Promise<{ content: string; patch?: CanvasPatch; patches?: CanvasPatch[]; isError?: boolean }> {
  1396→  const tool = tools.find((t) => t.name === toolName);
  1397→  if (!tool) {
  1398→    return { content: `Unknown tool: ${toolName}`, isError: true };
  1399→  }
  1400→  try {
  1401→    const result = await tool.execute(
  1402→      `call-${Date.now()}-${Math.random().toString(36).slice(2, 8)}`,
  1403→      args,
  1404→      undefined,
  1405→      undefined,
  1406→      undefined as any,
  1407→    );
  1408→    const text = result.content.map((c: any) => c.text ?? '').join('\n');
  1409→    const patch = (result.details as any)?.patch as CanvasPatch | undefined;
  1410→    const isError = (result as any).isError === true;
  1411→    return { content: text, patch, isError };
  1412→  } catch (err: any) {
  1413→    return { content: `Tool execution failed: ${err.message}`, isError: true };
  1414→  }
  1415→}

  1604→function buildUserFlow(flow: string, ox: number, oy: number): UserFlowResult {
  1605→  // Each step is a mobile screen (375 wide) with a 64px gap.
  1606→  const SCREEN_W = 375;
  1607→  const GAP = 80;
  1608→  const ARROW_COLOR = '#94a3b8';
  1609→
  1610→  const flows: Record<string, string[]> = {
  1611→    onboarding: ['mobile_login', 'mobile_signup', 'mobile_dashboard'],
  1612→    ecommerce: ['web_landing', 'mobile_dashboard', 'mobile_dashboard', 'mobile_login'],
  1613→    auth: ['mobile_login', 'mobile_dashboard', 'mobile_dashboard'],
  1614→    signup_funnel: ['web_landing', 'mobile_signup', 'mobile_login', 'mobile_dashboard'],
  1615→  };
  1616→  const screens = flows[flow] ?? flows.onboarding;
  1617→
  1618→  const frameIds: string[] = [];
  1619→  const shapes: Array<Partial<Shape> & { id: string }> = [];
  1620→
  1621→  screens.forEach((tmpl, i) => {
  1622→    const sx = ox + i * (SCREEN_W + GAP);
  1623→    const wf = buildWireframe(tmpl, sx, oy);
  1624→    // Rename the frame to include the step number.
  1625→    wf.shapes[0] = { ...wf.shapes[0], name: `Step ${i + 1}` };
  1626→    shapes.push(...wf.shapes);
  1627→    frameIds.push(wf.frameId);
  1628→
  1629→    // Add an arrow connector to the next screen.
  1630→    if (i < screens.length - 1) {
  1631→      const arrowStartX = sx + SCREEN_W;
  1632→      const arrowEndX = sx + SCREEN_W + GAP;
  1633→      const arrowY = oy + 300;
  1634→      shapes.push({
  1635→        id: crypto.randomUUID(),
  1636→        type: 'line',
  1637→        name: `Arrow ${i + 1}→${i + 2}`,
  1638→        x: arrowStartX,
  1639→        y: arrowY,
  1640→        width: arrowEndX - arrowStartX,
  1641→        height: 0,
  1642→        fill: ARROW_COLOR,
  1643→        stroke: ARROW_COLOR,
  1644→        strokeWidth: 2,
  1645→        radius: 0,
  1646→        fontSize: 14,
  1647→        textColor: ARROW_COLOR,
  1648→      });
  1649→    }
  1650→  });
  1651→  return { frameIds, shapes };
  1652→}

  1658→function buildDiagram(template: string, labels: string[], ox: number, oy: number): DiagramResult {
  1659→  const shapes: Array<Partial<Shape> & { id: string }> = [];
  1660→  const NODE_W = 160;
  1661→  const NODE_H = 56;
  1662→  const NODE_FILL = '#ffffff';
  1663→  const NODE_STROKE = '#475569';
  1664→  const ARROW = '#94a3b8';
  1665→
  1666→  const addNode = (label: string, x: number, y: number) => {
  1667→    shapes.push({
  1668→      id: crypto.randomUUID(),
  1669→      type: 'rectangle',

  1742→function capitalize(s: string): string {
  1743→  return s.charAt(0).toUpperCase() + s.slice(1);

  1746→function luminance(hex: string): number {
  1747→  const m = hex.replace('#', '');
  1748→  const r = parseInt(m.slice(0, 2), 16) / 255;
  1749→  const g = parseInt(m.slice(2, 4), 16) / 255;
  1750→  const b = parseInt(m.slice(4, 6), 16) / 255;
  1751→  const lin = (c: number) => (c <= 0.03928 ? c / 12.92 : Math.pow((c + 0.055) / 1.055, 2.4));
  1752→  return 0.2126 * lin(r) + 0.7152 * lin(g) + 0.0722 * lin(b);
  1753→}
  1754→
  1755→function contrastRatio(fg: string, bg: string): number {
  1756→  const l1 = luminance(fg);
  1757→  const l2 = luminance(bg);
  1758→  const lighter = Math.max(l1, l2);
  1759→  const darker = Math.min(l1, l2);
  1760→  return (lighter + 0.05) / (darker + 0.05);
  1761→}
  1762→
  1763→function isHighContrast(a: string, b: string): boolean {
  1764→  return contrastRatio(a, b) > 4.5;
  1765→}
