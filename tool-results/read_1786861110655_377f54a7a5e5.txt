  1382→// ---- Convert Pi tools → OpenAI tool spec -----------------------------------
  1383→//
  1384→// The Pi Agent SDK normally drives tools through `pi-ai` provider adapters.
  1385→// In this sandbox we don't have Anthropic/OpenAI keys, so we run the agent
  1386→// via `z-ai-web-dev-sdk` (which speaks the OpenAI tool-calling format).
  1387→// This helper converts our TypeBox-defined Pi tools into the JSON-schema
  1388→// tool spec that the LLM expects.
  1389→
  1390→import { Value } from '@sinclair/typebox/value';
  1391→
  1392→export interface OpenAIToolSpec {
  1393→  type: 'function';
  1394→  function: {
  1395→    name: string;
  1396→    description: string;
  1397→    parameters: object;
  1398→  };
  1399→}
  1400→
  1401→export function toolsToOpenAISpec(tools: ReturnType<typeof createCanvasTools>): OpenAIToolSpec[] {
  1402→  return tools.map((t) => ({
  1403→    type: 'function',
  1404→    function: {
  1405→      name: t.name,
  1406→      description: t.description,
  1407→      // TypeBox schemas ARE JSON-schema, so we can pass them through after
  1408→      // cleaning up any TypeBox-specific symbols.
  1409→      parameters: Value.Clean(t.parameters, {}) as object,
  1410→    },
  1411→  }));
  1412→}
  1413→
  1414→// ---- Execute a tool by name -------------------------------------------------
  1415→
  1416→export async function executeTool(
  1417→  tools: ReturnType<typeof createCanvasTools>,
  1418→  toolName: string,
  1419→  args: any,
  1420→): Promise<{ content: string; patch?: CanvasPatch; patches?: CanvasPatch[]; isError?: boolean }> {
  1421→  const tool = tools.find((t) => t.name === toolName);
  1422→  if (!tool) {
  1423→    return { content: `Unknown tool: ${toolName}`, isError: true };
  1424→  }
  1425→  try {
  1426→    const result = await tool.execute(
  1427→      `call-${Date.now()}-${Math.random().toString(36).slice(2, 8)}`,
  1428→      args,
  1429→      undefined,
  1430→      undefined,
  1431→      undefined as any,
  1432→    );
  1433→    const text = result.content.map((c: any) => c.text ?? '').join('\n');
  1434→    const patch = (result.details as any)?.patch as CanvasPatch | undefined;
  1435→    const isError = (result as any).isError === true;
  1436→    return { content: text, patch, isError };
  1437→  } catch (err: any) {
  1438→    return { content: `Tool execution failed: ${err.message}`, isError: true };
  1439→  }
  1440→}

  1629→function buildUserFlow(flow: string, ox: number, oy: number): UserFlowResult {
  1630→  // Each step is a mobile screen (375 wide) with a 64px gap.
  1631→  const SCREEN_W = 375;
  1632→  const GAP = 80;
  1633→  const ARROW_COLOR = '#94a3b8';
  1634→
  1635→  const flows: Record<string, string[]> = {
  1636→    onboarding: ['mobile_login', 'mobile_signup', 'mobile_dashboard'],
  1637→    ecommerce: ['web_landing', 'mobile_dashboard', 'mobile_dashboard', 'mobile_login'],
  1638→    auth: ['mobile_login', 'mobile_dashboard', 'mobile_dashboard'],
  1639→    signup_funnel: ['web_landing', 'mobile_signup', 'mobile_login', 'mobile_dashboard'],
  1640→  };
  1641→  const screens = flows[flow] ?? flows.onboarding;
  1642→
  1643→  const frameIds: string[] = [];
  1644→  const shapes: Array<Partial<Shape> & { id: string }> = [];
  1645→
  1646→  screens.forEach((tmpl, i) => {
  1647→    const sx = ox + i * (SCREEN_W + GAP);
  1648→    const wf = buildWireframe(tmpl, sx, oy);
  1649→    // Rename the frame to include the step number.
  1650→    wf.shapes[0] = { ...wf.shapes[0], name: `Step ${i + 1}` };
  1651→    shapes.push(...wf.shapes);
  1652→    frameIds.push(wf.frameId);
  1653→
  1654→    // Add an arrow connector to the next screen.
  1655→    if (i < screens.length - 1) {
  1656→      const arrowStartX = sx + SCREEN_W;
  1657→      const arrowEndX = sx + SCREEN_W + GAP;
  1658→      const arrowY = oy + 300;
  1659→      shapes.push({
  1660→        id: crypto.randomUUID(),
  1661→        type: 'line',
  1662→        name: `Arrow ${i + 1}→${i + 2}`,
  1663→        x: arrowStartX,
  1664→        y: arrowY,
  1665→        width: arrowEndX - arrowStartX,
  1666→        height: 0,
  1667→        fill: ARROW_COLOR,
  1668→        stroke: ARROW_COLOR,
  1669→        strokeWidth: 2,
  1670→        radius: 0,
  1671→        fontSize: 14,
  1672→        textColor: ARROW_COLOR,
  1673→      });
  1674→    }
  1675→  });
  1676→  return { frameIds, shapes };
  1677→}

  1683→function buildDiagram(template: string, labels: string[], ox: number, oy: number): DiagramResult {
  1684→  const shapes: Array<Partial<Shape> & { id: string }> = [];
  1685→  const NODE_W = 160;
  1686→  const NODE_H = 56;
  1687→  const NODE_FILL = '#ffffff';
  1688→  const NODE_STROKE = '#475569';
  1689→  const ARROW = '#94a3b8';
  1690→
  1691→  const addNode = (label: string, x: number, y: number) => {
  1692→    shapes.push({
  1693→      id: crypto.randomUUID(),
  1694→      type: 'rectangle',
  1695→      name: label,
  1696→      x, y, width: NODE_W, height: NODE_H,
  1697→      fill: NODE_FILL, stroke: NODE_STROKE, strokeWidth: 1, radius: 8,
  1698→      fontSize: 13, textColor: '#0f172a',
  1699→    });
  1700→    shapes.push({
  1701→      id: crypto.randomUUID(),
  1702→      type: 'text',
  1703→      name: `${label} label`,
  1704→      x: x + 12, y: y + 18, width: NODE_W - 24, height: 20,
  1705→      fill: 'transparent', text: label, fontSize: 13, textColor: '#0f172a',
  1706→      stroke: 'transparent', strokeWidth: 0, radius: 0,
  1707→    });
  1708→  };
  1709→  const addArrow = (x1: number, y1: number, x2: number, y2: number) => {
  1710→    shapes.push({
  1711→      id: crypto.randomUUID(),
  1712→      type: 'line',
  1713→      name: 'connector',
  1714→      x: x1, y: y1, width: x2 - x1, height: y2 - y1,
  1715→      fill: ARROW, stroke: ARROW, strokeWidth: 2, radius: 0,
  1716→      fontSize: 14, textColor: ARROW,
  1717→    });
  1718→  };
  1719→
  1720→  if (template === 'flowchart') {
  1721→    // Top-down: each node centered, vertical arrow between.
  1722→    const cx = ox + 200;
  1723→    let cy = oy;
  1724→    const STEP = NODE_H + 60;
  1725→    labels.forEach((label, i) => {
  1726→      addNode(label, cx, cy);
  1727→      if (i < labels.length - 1) {
  1728→        addArrow(cx + NODE_W / 2, cy + NODE_H, cx + NODE_W / 2, cy + STEP);
  1729→      }
  1730→      cy += STEP;
  1731→    });
  1732→  } else {
  1733→    // Mindmap: central node + radial children.
  1734→    const cx = ox + 360;
  1735→    const cy = oy + 280;
  1736→    const centerLabel = labels[0];
  1737→    addNode(centerLabel, cx - NODE_W / 2, cy - NODE_H / 2);
  1738→    const radius = 220;
  1739→    const childCount = labels.length - 1;
  1740→    for (let i = 1; i < labels.length; i++) {
  1741→      const angle = (i - 1) / Math.max(1, childCount) * Math.PI * 2;
  1742→      const px = cx + Math.cos(angle) * radius - NODE_W / 2;
  1743→      const py = cy + Math.sin(angle) * radius - NODE_H / 2;
  1744→      addArrow(cx, cy, px + NODE_W / 2, py + NODE_H / 2);
  1745→      addNode(labels[i], px, py);
  1746→    }
  1747→  }
  1748→  return { shapes };
  1749→}

  1771→function luminance(hex: string): number {
  1772→  const m = hex.replace('#', '');
  1773→  const r = parseInt(m.slice(0, 2), 16) / 255;
  1774→  const g = parseInt(m.slice(2, 4), 16) / 255;
  1775→  const b = parseInt(m.slice(4, 6), 16) / 255;
  1776→  const lin = (c: number) => (c <= 0.03928 ? c / 12.92 : Math.pow((c + 0.055) / 1.055, 2.4));
  1777→  return 0.2126 * lin(r) + 0.7152 * lin(g) + 0.0722 * lin(b);
  1778→}
  1779→
  1780→function contrastRatio(fg: string, bg: string): number {
  1781→  const l1 = luminance(fg);
  1782→  const l2 = luminance(bg);
  1783→  const lighter = Math.max(l1, l2);
  1784→  const darker = Math.min(l1, l2);
  1785→  return (lighter + 0.05) / (darker + 0.05);
  1786→}
  1787→
  1788→function isHighContrast(a: string, b: string): boolean {
  1789→  return contrastRatio(a, b) > 4.5;
  1790→}
